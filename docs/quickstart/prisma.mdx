## Setup Prisma

Initialize Prisma in your project:

```bash
npx prisma init
```

Define your schema in `prisma/schema.prisma`:

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Product {
  id        String   @id @default(uuid())
  name      String
  price     Float
  stock     Int
  status    String
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}
```

Generate Prisma Client and push schema:

```bash
npx prisma generate
npx prisma db push
```

## Define Your Aggregate

Create your domain model:

```typescript
import { z } from "zod";
import { Aggregate, EntityValidation, Id } from "@woltz/rich-domain";

const productSchema = z.object({
  id: z.custom<Id>((val) => val instanceof Id),
  name: z.string().min(3),
  price: z.number().positive(),
  stock: z.number().int().min(0),
  status: z.enum(["active", "inactive"]),
});

type ProductProps = z.infer<typeof productSchema>;

class Product extends Aggregate<ProductProps> {
  protected static validation: EntityValidation<ProductProps> = {
    schema: productSchema,
    config: {
      onCreate: true,
      onUpdate: true,
      throwOnError: true,
    },
  };

  get name() {
    return this.props.name;
  }

  get price() {
    return this.props.price;
  }

  get stock() {
    return this.props.stock;
  }

  updatePrice(newPrice: number) {
    this.props.price = newPrice;
  }

  decreaseStock(amount: number) {
    this.props.stock -= amount;
  }

  activate() {
    this.props.status = "active";
  }

  deactivate() {
    this.props.status = "inactive";
  }
}
```

## Create Mappers

### ToDomain Mapper

Maps from Prisma to Domain:

```typescript
import { Mapper, Id } from "@woltz/rich-domain";
import { ProductSchema } from "../schemas/product.schema";

export class ProductToDomainMapper extends Mapper<ProductSchema, Product> {
  build(raw: ProductSchema): Product {
    return new Product({
      id: Id.from(raw.id),
      name: raw.name,
      price: raw.price,
      stock: raw.stock,
      status: raw.status as "active" | "inactive",
    });
  }
}
```

### ToPersistence Mapper

Maps from Domain to Prisma using `PrismaToPersistence`:

```typescript
import { PrismaClient } from "@prisma/client";
import { 
  PrismaToPersistence, 
  PrismaBatchExecutor 
} from "@woltz/rich-domain-prisma";
import { AggregateChanges, EntitySchemaRegistry } from "@woltz/rich-domain";

export class ProductToPersistenceMapper extends PrismaToPersistence<
  Product,
  PrismaClient
> {
  protected readonly registry = new EntitySchemaRegistry().register({
    entity: "Product",
    table: "product",
    // Optional: Map domain fields to database columns
    // fields: {
    //   content: "main_content", // domain.content -> db.main_content
    // },
  });

  protected async onCreate(product: Product): Promise<void> {
    await this.context.product.create({
      data: {
        id: product.id.value,
        name: product.name,
        price: product.price,
        stock: product.stock,
        status: product.status,
        createdAt: new Date(),
        updatedAt: new Date(),
      },
    });
  }

  protected async onUpdate(
    product: Product,
    changes: AggregateChanges
  ): Promise<void> {
    const executor = new PrismaBatchExecutor(this.context, {
      registry: this.registry,
      rootId: product.id.value,
       // Custom data mappers per entity. 
       // Use this to transform domain objects to persistence format. (optional)
      dataMappers: {
        Produtct: (item) => {
          (...)
        }
      }
    });

    // Automatically creates UPDATE queries based on changed fields
    await executor.execute(changes);
  }
}
```

<Note>
  The `EntitySchemaRegistry` helps `PrismaBatchExecutor` understand your domain structure and automatically generate the correct SQL operations for creates, updates, and deletes.
</Note>

## Create Repository

Implement the repository:

```typescript
import { PrismaRepository, PrismaUnitOfWork } from "@woltz/rich-domain-prisma";
import { PrismaClient, Prisma } from "@prisma/client";

const prisma = new PrismaClient();
const uow = new PrismaUnitOfWork(prisma);

interface IProductRepository {
  save(product: Product): Promise<void>;
  findById(id: string): Promise<Product | null>;
}

class ProductRepository
  extends PrismaRepository<Product, ProductSchema, PrismaClient>
  implements IProductRepository
{
  constructor() {
    super(
      new ProductToPersistenceMapper(prisma, uow),
      new ProductToDomainMapper(),
      prisma,
      uow
    );
  }

  protected get model() {
    return "product";
  }

  protected generateSearchQuery(search: string) {
    return [
      { name: { contains: search, mode: "insensitive" } }
    ] satisfies Prisma.ProductWhereInput[];
  }

  async findById(id: string): Promise<Product | null> {
    const data = await this.modelAccessor.findUnique({ 
      where: { id } 
    });
    
    if (!data) return null;
    return this.toDomainMapper.build(data);
  }
}
```

## Use It

```typescript
const repository = new ProductRepository();

// Create product
const product = new Product({
  name: "Mechanical Keyboard",
  price: 149.99,
  stock: 50,
  status: "active",
});

// Save to database
await repository.save(product);
console.log("Product created:", product.id.value);

// Update
product.updatePrice(129.99);
product.decreaseStock(5);
await repository.save(product);

// Query
const found = await repository.findById(product.id.value);
console.log("Found product:", found?.name); // "Mechanical Keyboard"

// Use Criteria for complex queries
const criteria = Criteria.create<Product>()
  .where("status", "equals", "active")
  .where("price", "greaterThan", 100)
  .orderBy("name", "asc")
  .paginate(1, 10);

const results = await repository.find(criteria);
console.log(results.data); // Array of products
console.log(results.meta); // Pagination info
```

## Working with Relations

The `PrismaBatchExecutor` automatically handles nested entities and collections:

```typescript
// Example with relations
const userSchema = z.object({
  id: z.custom<Id>((val) => val instanceof Id),
  email: z.string().email(),
  name: z.string(),
  posts: z.array(z.custom<Post>()),
});

class User extends Aggregate<UserProps> {
  // ... validation

  addPost(post: Post) {
    this.props.posts.push(post);
  }

  removePost(postId: string) {
    const index = this.props.posts.findIndex(p => p.id.value === postId);
    if (index !== -1) {
      this.props.posts.splice(index, 1);
    }
  }
}

// ToPersistence with relations
export class UserToPersistenceMapper extends PrismaToPersistence<
  User,
  PrismaClient
> {
  protected readonly registry = new EntitySchemaRegistry()
    .register({
      entity: "User",
      table: "user",
    })
    .register({
      entity: "Post",
      table: "post",
      parentFk: {
        field: "authorId",
        parentEntity: "User",
      },
    });

  protected async onCreate(user: User): Promise<void> {
    await this.context.user.create({
      data: {
        id: user.id.value,
        email: user.email,
        name: user.name,
        createdAt: new Date(),
        updatedAt: new Date(),
        posts: {
          createMany: user.posts.length
            ? {
                data: user.posts.map((post) => ({
                  id: post.id.value,
                  title: post.title,
                  content: post.content,
                  published: post.published,
                  authorId: user.id.value,
                  createdAt: new Date(),
                  updatedAt: new Date(),
                })),
              }
            : undefined,
        },
      },
    });
  }

  protected async onUpdate(
    user: User,
    changes: AggregateChanges
  ): Promise<void> {
    const executor = new PrismaBatchExecutor(this.context, {
      registry: this.registry,
      rootId: user.id.value,
    });

    // Automatically handles:
    // - Field updates on User
    // - Post creates/updates/deletes
    // - Nested entity changes
    await executor.execute(changes);
  }
}

// Usage
const user = User.create({ email: "john@example.com", name: "John" });

const post = Post.create({
  title: "My First Post",
  content: "Hello World",
  authorId: user.id.value,
});

user.addPost(post);

await userRepository.save(user);
// Both User and Post are created in a single transaction
```

## Many-to-Many Relations

Handle N:N relationships with junction tables:

```typescript
// Prisma Schema
model Post {
  id       String     @id
  title    String
  tagPosts TagPost[]
}

model Tag {
  id       String     @id
  name     String
  tagPosts TagPost[]
}

model TagPost {
  postId String
  tagId  String
  post   Post   @relation(fields: [postId], references: [id])
  tag    Tag    @relation(fields: [tagId], references: [id])

  @@id([postId, tagId])
}
```

```typescript
// Domain
class Post extends Aggregate<PostProps> {
  addTag(tag: Tag) {
    if (!this.props.tags.some(t => t.id.equals(tag.id))) {
      this.props.tags.push(tag);
    }
  }

  removeTag(tagId: Id) {
    const index = this.props.tags.findIndex(t => t.id.equals(tagId));
    if (index !== -1) {
      this.props.tags.splice(index, 1);
    }
  }
}

// ToPersistence
export class PostToPersistenceMapper extends PrismaToPersistence<
  Post,
  PrismaClient
> {
  protected readonly registry = new EntitySchemaRegistry().register({
    entity: "Post",
    table: "post",
    collections: {
      tags: {
        type: "reference",
        entity: "Tag",
        // Junction config is necessary, as we created the pivot table manually
        junction: {
          table: "TagPost", // <- The pivot table name, in example name is TagPost
          sourceKey: "postId",
          targetKey: "tagId",
        },
      },
    },
  });

  protected async onCreate(post: Post): Promise<void> {
    await this.context.post.create({
      data: {
        id: post.id.value,
        title: post.title,
        content: post.content,
        published: post.published,
      },
    });

    // Create junction records
    if (post.tags.length) {
      await this.context.tagPost.createMany({
        data: post.tags.map((tag) => ({
          tagId: tag.id.value,
          postId: post.id.value,
        })),
      });
    }
  }

  protected async onUpdate(
    post: Post,
    changes: AggregateChanges
  ): Promise<void> {
    const executor = new PrismaBatchExecutor(this.context, {
      registry: this.registry,
      rootId: post.id.value,
    });

    // Automatically handles tag additions/removals in junction table
    await executor.execute(changes);
  }
}
```

<Note>
  A collection junction is only necessary when you remove the responsibility of creating the pivot table from Prisma and create it manually.
</Note>
