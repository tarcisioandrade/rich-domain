export const FilterDemo = () => {
  const [filters, setFilters] = useState([]);
  const [isAddingFilter, setIsAddingFilter] = useState(false);
  const [selectedField, setSelectedField] = useState(null);
  const [selectedOperator, setSelectedOperator] = useState("equals");
  const [filterValue, setFilterValue] = useState("");

  const fields = [
    { 
      field: "name", 
      label: "Name", 
      type: "string",
      operators: ["equals", "contains", "startsWith", "endsWith", "isNull"]
    },
    { 
      field: "status", 
      label: "Status", 
      type: "string",
      operators: ["equals", "notEquals", "in", "isNull"],
      options: ["active", "inactive", "pending"]
    },
    { 
      field: "price", 
      label: "Price", 
      type: "number",
      operators: ["equals", "greaterThan", "lessThan", "between"]
    },
    { 
      field: "createdAt", 
      label: "Created At", 
      type: "date",
      operators: ["equals", "greaterThan", "lessThan", "between"]
    },
  ];

  const operatorLabels = {
    equals: "equals",
    notEquals: "not equals",
    contains: "contains",
    startsWith: "starts with",
    endsWith: "ends with",
    greaterThan: ">",
    lessThan: "<",
    between: "between",
    in: "in",
    isNull: "is empty",
  };

  const usedFields = filters.map(f => f.field);
  const availableFields = fields.filter(f => !usedFields.includes(f.field));

  const handleAddFilter = () => {
    if (!selectedField || (!filterValue && selectedOperator !== "isNull")) return;
    
    setFilters([...filters, {
      field: selectedField.field,
      label: selectedField.label,
      operator: selectedOperator,
      value: filterValue,
    }]);
    
    setIsAddingFilter(false);
    setSelectedField(null);
    setSelectedOperator("equals");
    setFilterValue("");
  };

  const handleRemoveFilter = (index) => {
    setFilters(filters.filter((_, i) => i !== index));
  };

  const handleClearAll = () => {
    setFilters([]);
  };

  const handleFieldSelect = (field) => {
    setSelectedField(field);
    setSelectedOperator(field.operators[0]);
    setFilterValue("");
  };

  return (
    <div className="w-full p-4 border border-zinc-200 dark:border-zinc-700 rounded-lg bg-white dark:bg-zinc-900">
      <div className="flex flex-wrap items-center gap-2">
        {/* Active Filters */}
        {filters.map((filter, index) => (
          <div 
            key={index}
            className="flex items-center gap-1.5 px-2 py-1 rounded-md border border-zinc-200 dark:border-zinc-700 bg-zinc-50 dark:bg-zinc-800 text-sm"
          >
            <span className="font-medium text-zinc-700 dark:text-zinc-300">{filter.label}</span>
            <span className="text-zinc-500 dark:text-zinc-400">{operatorLabels[filter.operator]}</span>
            {filter.operator !== "isNull" && (
              <span className="px-1.5 py-0.5 rounded bg-indigo-100 dark:bg-indigo-900 text-indigo-700 dark:text-indigo-300 font-medium">
                {filter.value}
              </span>
            )}
            <button
              onClick={() => handleRemoveFilter(index)}
              className="ml-1 p-0.5 rounded hover:bg-zinc-200 dark:hover:bg-zinc-700 text-zinc-500 hover:text-zinc-700 dark:hover:text-zinc-300"
            >
              <svg className="w-3.5 h-3.5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
              </svg>
            </button>
          </div>
        ))}

        {/* Clear All Button */}
        {filters.length > 0 && (
          <button
            onClick={handleClearAll}
            className="flex items-center gap-1 px-2 py-1 rounded-md border border-zinc-200 dark:border-zinc-700 hover:bg-zinc-100 dark:hover:bg-zinc-800 text-sm text-zinc-600 dark:text-zinc-400"
          >
            <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
            </svg>
          </button>
        )}

        {/* Add Filter Button / Form */}
        {!isAddingFilter ? (
          <button
            onClick={() => setIsAddingFilter(true)}
            disabled={availableFields.length === 0}
            className="flex items-center gap-1.5 px-3 py-1.5 rounded-md border border-zinc-200 dark:border-zinc-700 hover:bg-zinc-100 dark:hover:bg-zinc-800 text-sm text-zinc-600 dark:text-zinc-400 disabled:opacity-50 disabled:cursor-not-allowed"
          >
            <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M3 4a1 1 0 011-1h16a1 1 0 011 1v2.586a1 1 0 01-.293.707l-6.414 6.414a1 1 0 00-.293.707V17l-4 4v-6.586a1 1 0 00-.293-.707L3.293 7.293A1 1 0 013 6.586V4z" />
            </svg>
            {filters.length === 0 ? "Add Filter" : "Add"}
          </button>
        ) : (
          <div className="flex items-center gap-2 p-2 rounded-md border border-indigo-200 dark:border-indigo-800 bg-indigo-50 dark:bg-indigo-950">
            {/* Field Selector */}
            <select
              value={selectedField?.field || ""}
              onChange={(e) => {
                const field = fields.find(f => f.field === e.target.value);
                if (field) handleFieldSelect(field);
              }}
              className="px-2 py-1 rounded border border-zinc-300 dark:border-zinc-600 bg-white dark:bg-zinc-800 text-sm text-zinc-700 dark:text-zinc-300"
            >
              <option value="">Select field...</option>
              {availableFields.map(field => (
                <option key={field.field} value={field.field}>{field.label}</option>
              ))}
            </select>

            {/* Operator Selector */}
            {selectedField && (
              <select
                value={selectedOperator}
                onChange={(e) => setSelectedOperator(e.target.value)}
                className="px-2 py-1 rounded border border-zinc-300 dark:border-zinc-600 bg-white dark:bg-zinc-800 text-sm text-zinc-700 dark:text-zinc-300"
              >
                {selectedField.operators.map(op => (
                  <option key={op} value={op}>{operatorLabels[op]}</option>
                ))}
              </select>
            )}

            {/* Value Input */}
            {selectedField && selectedOperator !== "isNull" && (
              selectedField.options ? (
                <select
                  value={filterValue}
                  onChange={(e) => setFilterValue(e.target.value)}
                  className="px-2 py-1 rounded border border-zinc-300 dark:border-zinc-600 bg-white dark:bg-zinc-800 text-sm text-zinc-700 dark:text-zinc-300"
                >
                  <option value="">Select value...</option>
                  {selectedField.options.map(opt => (
                    <option key={opt} value={opt}>{opt}</option>
                  ))}
                </select>
              ) : (
                <input
                  type={selectedField.type === "number" ? "number" : selectedField.type === "date" ? "date" : "text"}
                  value={filterValue}
                  onChange={(e) => setFilterValue(e.target.value)}
                  placeholder="Enter value..."
                  className="px-2 py-1 rounded border border-zinc-300 dark:border-zinc-600 bg-white dark:bg-zinc-800 text-sm text-zinc-700 dark:text-zinc-300 w-32"
                />
              )
            )}

            {/* Action Buttons */}
            <button
              onClick={handleAddFilter}
              disabled={!selectedField || (!filterValue && selectedOperator !== "isNull")}
              className="px-2 py-1 rounded bg-indigo-600 hover:bg-indigo-700 text-white text-sm font-medium disabled:opacity-50 disabled:cursor-not-allowed"
            >
              Add
            </button>
            <button
              onClick={() => {
                setIsAddingFilter(false);
                setSelectedField(null);
                setFilterValue("");
              }}
              className="px-2 py-1 rounded border border-zinc-300 dark:border-zinc-600 hover:bg-zinc-100 dark:hover:bg-zinc-700 text-sm text-zinc-600 dark:text-zinc-400"
            >
              Cancel
            </button>
          </div>
        )}
      </div>

      {/* Generated Criteria Preview */}
      {filters.length > 0 && (
        <div className="mt-4 p-3 rounded-md bg-zinc-100 dark:bg-zinc-800 border border-zinc-200 dark:border-zinc-700">
          <div className="text-xs font-medium text-zinc-500 dark:text-zinc-400 mb-2">Generated Criteria:</div>
          <pre className="text-xs text-zinc-700 dark:text-zinc-300 overflow-x-auto">
{`Criteria.create<User>()
${filters.map(f => {
  if (f.operator === "isNull") {
    return `  .whereNull("${f.field}")`;
  }
  const value = typeof f.value === "string" && isNaN(f.value) ? `"${f.value}"` : f.value;
  return `  .where("${f.field}", "${f.operator}", ${value})`;
}).join('\n')}`}
          </pre>
        </div>
      )}
    </div>
  );
};

<FilterDemo />
